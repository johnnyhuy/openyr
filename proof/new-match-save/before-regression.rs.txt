
#[test]
fn new_front_match_does_not_overwrite_an_earlier_named_save_on_f5() {
    let (mut app, _, _) = scene((640, 480));
    let directory = PreferenceDirectory::new();
    let named = directory.0.join("Earlier campaign.oyrsave");
    yr_app::snapshot::save(app.session.as_ref().unwrap(), &named).unwrap();
    let before = std::fs::read(&named).unwrap();
    app.save_path = named.clone();
    app.apply_menu_effect(yr_app::menu::Effect::MainMenu);
    assert!(app.session.is_none() && app.menu.front);
    let (job, sender, _) = crate::menu_flow::LoadJob::channel(Page::Skirmish);
    app.load_job = Some(job);
    app.open_menu(Page::Loading);
    let fresh = session();
    assert_ne!(yr_app::snapshot::encode(&fresh).unwrap(), before);
    sender.send(Ok(crate::menu_flow::Prepared::Match {
        session: Box::new(fresh),
        map: yr_render::RenderedMap { width: 640, height: 480, rgba: Vec::new(),
            origin_x: 0, origin_y: 0, scale: 1, summary: "authored fresh match".into() },
        map_name: "NEW.MAP".into(), save_path: None, sprites: None,
    })).ok().unwrap();
    app.poll_menu_load();
    assert_eq!(app.menu.page, Page::Closed);
    assert!(!app.menu.front);
    key(&mut app, KeyCode::F5, ElementState::Pressed);
    assert!(std::fs::read(&named).unwrap() == before,
        "a new match's quicksave replaced the earlier named match");
}
